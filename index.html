function downloadInvoicePDF() { let element = document.getElementById('printableInvoiceArea'); let invNo = document.getElementById('invNumber').value.replace(/\//g, '_'); let opt = { margin: 5, filename: `Invoice_${invNo}.pdf`, image: { type: 'jpeg', quality: 0.98 }, html2canvas: { scale: 2, useCORS: true, scrollY: 0, windowWidth: element.scrollWidth, windowHeight: element.scrollHeight }, jsPDF: { unit: 'mm', format: 'a4', orientation: 'portrait' } }; html2pdf().from(element).set(opt).save(); } async function sharePDFDirectly() { let element = document.getElementById('printableInvoiceArea'); let invNo = document.getElementById('invNumber').value.replace(/\//g, '_'); let opt = { margin: 5, filename: `Invoice_${invNo}.pdf`, image: { type: 'jpeg', quality: 0.98 }, html2canvas: { scale: 2, useCORS: true, scrollY: 0, windowWidth: element.scrollWidth, windowHeight: element.scrollHeight }, jsPDF: { unit: 'mm', format: 'a4', orientation: 'portrait' } }; if (navigator.share) { try { let worker = html2pdf().from(element).set(opt); let pdfBlob = await worker.outputPdf('blob'); let file = new File([pdfBlob], `Invoice_${invNo}.pdf`, { type: 'application/pdf' }); if (navigator.canShare && navigator.canShare({ files: [file] })) { await navigator.share({ title: `Tax Invoice ${invNo}`, text: 'Here is your tax invoice PDF.', files: [file] }); } else { alert("Direct sharing not supported on this browser. Downloading PDF instead."); worker.save(); } } catch (error) { console.log("Sharing error:", error); } } else { alert("Direct sharing is not supported on this browser. Use 'Download PDF' instead."); downloadInvoicePDF(); } }
